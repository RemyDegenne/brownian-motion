\section{Basic definitions}

First, recall the definitions of a filtration, an adapted process, a (sub)martingale, a stopping time and a stopped process, which are already in Mathlib.


\begin{definition}[Filtration]\label{def:filtration}
  \mathlibok
  \lean{MeasureTheory.Filtration}
A filtration on a measurable space $(\Omega, \mathcal{A})$ with measure $P$ indexed by a preordered set $T$ is a family of sigma-algebras $\mathcal{F} = (\mathcal{F}_t)_{t \in T}$ such that for all $i \le j$, $\mathcal{F}_i \subseteq \mathcal{F}_j$ and for all $t \in T$, $\mathcal{F}_t \subseteq \mathcal{A}$.
\end{definition}


\begin{definition}\label{def:adapted}
  \uses{def:filtration}
  \mathlibok
  \lean{MeasureTheory.Adapted}
A process $X : T \to \Omega \to E$ is said to be adapted with respect to a filtration $\mathcal{F}$ if for all $t \in T$, $X_t$ is $\mathcal{F}_t$-measurable.
\end{definition}


\begin{definition}[Progressively measurable]\label{def:IsStronglyProgressive}
  \uses{def:filtration}
  \mathlibok
  \lean{MeasureTheory.IsStronglyProgressive}
A stochastic process $X$ is said to be progressively measurable with respect to a filtration $\mathcal{F}$ if at each point in time $i$, $X$ restricted to $(-\infty, i] \times \Omega$ is measurable with respect to the product $\sigma$-algebra where the $\sigma$-algebra used for $\Omega$ is $\mathcal{F}_i$.
\end{definition}

\begin{definition}[Right continuous function]\label{def:RightContinuous}
  \leanok
  \lean{IsRightContinuous}
  Let $T$ be equipped with a partial order and $f : T \to E$ be a function. Then $f$ is right
  continuous if for all $t \in T$, $\lim_{s \downarrow t} f(s) = f(t)$.
\end{definition}

\begin{definition}[Cadlag function]\label{def:IsCadlag}
  \uses{def:RightContinuous}
  \leanok
  \lean{IsCadlag}
  A function $f : T \to E$ is cadlag if it is right continuous and the limit $\lim_{s \uparrow t} f(s)$ exists for all $t \in T$.
\end{definition}


\begin{lemma}\label{lem:Adapted.isStronglyProgressive_of_rightContinuous}
  \uses{def:adapted, def:IsStronglyProgressive, def:RightContinuous}
  \leanok
  \lean{MeasureTheory.StronglyAdapted.isStronglyProgressive_of_rightContinuous}
Suppose $T$ is a linearly ordered set equipped with the order topology. If $T$ is second countable and a stochastic process $(X_t)_{t \in T}$ is right continuous and adapted, then it is progressively measurable.
\end{lemma}

\begin{proof}
  \leanok
  Fixing $t \in T$, let $S$ be a union of a dense set of $(-\infty,t]$ and the set of points isolated from right in $(-\infty,t]$.
  As $T$ is second countable, $S$ can be chosen so that it is countable, which means that there exists a surjective function $u: \mathbb{N} \rightarrow S$.
  We need to show that $X$ restricted to $(-\infty, t] \times \Omega$ is measurable with respect to $\mathcal{B}((-\infty, t]) \otimes \mathcal{F}_t$.
  To this end, we rearrange the image of the first $n$ numbers under the map $u$ in nondecreasing order, denoted by $v_0\le v_1\le \cdots\le v_n$,
  and define a left continuous discrete approximation of $X$ on $(-\infty, t]$ by
  $$X^n_s = \begin{cases}
    X_{v_0} & \text{if } s \le v_0 \\
    X_{v_i} & \text{if } s \in (- v_{i-1}, v_i]
  \end{cases}$$
  As $X$ is right continuous, it is easy to see that $X^n \to X$ pointwise as $n \to \infty$.
  Thus, as each $X^n$ is measurable, it follows that $X$ is also measurable (by using \verb|stronglyMeasurable_of_tendsto|).
\end{proof}


\begin{definition}[Martingale]\label{def:Martingale}
  \uses{def:adapted}
  \mathlibok
  \lean{MeasureTheory.Martingale}
Let $\mathcal{F}$ be a filtration on a measurable space $\Omega$ with measure $P$ indexed by $T$.
A family of functions $M : T \to \Omega \to E$ is a martingale with respect to a filtration $\mathcal{F}$ if $M$ is adapted with respect to $\mathcal{F}$ and for all $i \le j$, $P[M_j \mid \mathcal{F}_i] = M_i$ almost surely.
\end{definition}


\begin{definition}[Submartingale]\label{def:Submartingale}
  \uses{def:adapted}
  \mathlibok
  \lean{MeasureTheory.Submartingale}
Let $\mathcal{F}$ be a filtration on a measurable space $\Omega$ with measure $P$ indexed by $T$.
A family of functions $M : T \to \Omega \to E$ is a submartingale with respect to a filtration $\mathcal{F}$ if $M$ is adapted with respect to $\mathcal{F}$ and for all $i \le j$, $P[M_j \mid \mathcal{F}_i] \ge M_i$ almost surely.
\end{definition}

\begin{lemma}\label{lem:Martingale.congr}
  \uses{def:Martingale, def:adapted}
  \leanok
  \lean{MeasureTheory.Martingale.congr}
If $X$ is a martingale and $Y$ is an adapted modification of $X$, then $Y$ is a martingale.
\end{lemma}

\begin{proof}\leanok
Let $i \le j$ in $T$. We want to show that $P[Y_j \mid \mathcal{F}_i] = Y_i$ almost surely.
It suffices to show that $\int_A Y_j \: dP = \int_A Y_i \: dP$ for all $A \in \mathcal{F}_i$.
Let then $A \in \mathcal{F}_i$.
\begin{align*}
  \int_A Y_j \: dP
  &= \int_A X_j \: dP
  = \int_A X_i \: dP
  = \int_A Y_i \: dP
  \: .
\end{align*}
\end{proof}


\begin{lemma}\label{lem:Submartingale.congr}
  \uses{def:Submartingale, def:adapted}
  \leanok
  \lean{MeasureTheory.Submartingale.congr}
If $X$ is a submartingale and $Y$ is an adapted modification of $X$, then $Y$ is a submartingale.
\end{lemma}

\begin{proof}\leanok
Let $i \le j$ in $T$. We want to show that $P[Y_j \mid \mathcal{F}_i] \ge Y_i$ almost surely.
It suffices to show that $\int_A Y_j \: dP \ge \int_A Y_i \: dP$ for all $A \in \mathcal{F}_i$.
Let then $A \in \mathcal{F}_i$.
\begin{align*}
  \int_A Y_j \: dP
  &= \int_A X_j \: dP
  \ge \int_A X_i \: dP
  = \int_A Y_i \: dP
  \: .
\end{align*}
\end{proof}


\begin{lemma}[Jensen's inequality for the conditional expectation]\label{lem:conditional_jensen}
  \mathlibok
  \lean{ConvexOn.map_condExp_le_univ}
Let $X : \Omega \to E$ be an integrable random variable with values in a normed space $E$ and let $\phi : E \to \mathbb{R}$ be a convex function such that $\phi \circ X$ is integrable.
Then, for any sub-$\sigma$-algebra $\mathcal{G}$, we have
\begin{align*}
  \phi\left( \mathbb{E}[X \mid \mathcal{G}] \right)
  &\le \mathbb{E}[\phi(X) \mid \mathcal{G}] \quad \text{a.s.}
\end{align*}
\end{lemma}

Done in a Mathlib PR for finite measures: \href{https://github.com/leanprover-community/mathlib4/pull/27953}{\#27953}.
\end{proof}

\begin{lemma}\label{lem:Martingale.submartingale_convex_comp}
  \uses{def:Martingale, def:Submartingale}
  \leanok
  \lean{MeasureTheory.Martingale.submartingale_convex_comp}
Let $X : T \rightarrow \Omega\rightarrow E$ a martingale with values in a normed space $E$.

\end{proof}

\begin{definition}[Stopping time]\label{def:IsStoppingTime}
  \uses{def:filtration}
  \mathlibok
  \lean{MeasureTheory.IsStoppingTime}
A stopping time with respect to some filtration $\mathcal{F}$ indexed by $T$ is a function $\tau : \Omega \to T \cup \{\infty\}$ such that for all $i$, the preimage of $\{j \mid j \le i\}$ along $\tau$ is measurable with respect to $\mathcal{F}_i$.
\end{definition}

  $\mathcal{F}_\tau = \bigcap_{t \in T} \{A \in \mathcal{F} \mid A \cap \{\tau \le t\} \in \mathcal{F}_t\}.$
\end{definition}

\begin{definition}[Stopped process]\label{def:stoppedProcess}
  \mathlibok
  \lean{MeasureTheory.stoppedProcess}
Let $X : T \to \Omega \to E$ be a stochastic process and let $\tau : \Omega \to T$.
The stopped process with respect to $\tau$ is defined by
\begin{align*}
  (X^{\tau})_t = \begin{cases}
    X_t & \text{if } t \le \tau \\
    X_{\tau} & \text{otherwise}

\end{definition}

\begin{definition}[Hitting time]\label{def:hittingAfter}
  \mathlibok
  \lean{MeasureTheory.hittingAfter}
For $X : T \to \Omega \to E$ a stochastic process, $B$ a subset of $E$ and $t_0 \in T$, the hitting time of $X$ in $B$ after $t_0$ is the random variable $\Omega \to T\cup\{\infty\}$ defined by
\begin{align*}
  \tau_{B, t_0}(\omega) = \inf\{t \in T \mid t \ge t_0, \: X_t(\omega) \in B\} \: ,
\end{align*}
in which the infimum is infinite if the set is empty.
\end{definition}

  \tau_{B, t_0, t_1}(\omega) = \inf\{t \in T \mid t_0 \le t \le t_1, \: X_t(\omega) \in B\} \: ,

in which the infimum is infinite if the set is empty.

\section{Right-continuous filtrations}

We now give the definition of a filtered probability space satisfying the usual conditions.


\begin{definition}[Left continuation]\label{def:leftLimitFiltration}
  \uses{def:filtration}
Assume that $T$ is a partial order. For $\mathcal{F}$ a filtration indexed by $T$ and $t \in T$, we define the \emph{left continuation} as
$$\mathcal{F}_{t-} =
  \begin{cases}
    \mathcal{F}_t, & \text{if $t$ is isolated on the left in the order topology}; \\
    \bigsqcup_{s < t} \mathcal{F}_s, & \text{otherwise}.
  \end{cases}
$$

Note that $\bigsqcup$ denotes the supremum in the lattice of sigma-algebras on $\Omega$.
\end{definition}


\begin{definition}[Right continuation]\label{def:rightLimitFiltration}
  \uses{def:filtration}
  \leanok
  \lean{MeasureTheory.Filtration.rightCont}
Assume that $T$ is a partial order. For $\mathcal{F}$ a filtration indexed by $T$ and $t \in T$, we define the \emph{right continuation} as
$$\mathcal{F}_{t+} =
  \begin{cases}
    \mathcal{F}_t, & \text{if $t$ is isolated on the right in the order topology}; \\
    \bigsqcap_{s > t} \mathcal{F}_s, & \text{otherwise}.
  \end{cases}
$$

Note that $\bigsqcap$ denotes the infimum in the lattice of sigma-algebras on $\Omega$.
\end{definition}


\begin{lemma}\label{lem:filtrationRightCont}
  \uses{def:filtration, def:rightLimitFiltration}
  \leanok
  \lean{MeasureTheory.Filtration.rightCont}
The right continuation of $\mathcal{F}$ is a filtration.
\end{lemma}

\begin{proof}\leanok
We endow $T$ with the order topology. Let us first prove that the right continuation is nondecreasing. Let $s, t \in T$ such that $s \le t$.
\begin{itemize}
  \item Suppose $s$ is isolated on the right. Then $\mathcal{F}_{s+} = \mathcal{F}_s$.
  \begin{itemize}
    \item If $t$ is isolated on the right, then $\mathcal{F}_{t+} = \mathcal{F}_t$. Because $\mathcal{F}$ is a filtration, we have $\mathcal{F}_s \subseteq \mathcal{F}_t$, and thus $\mathcal{F}_{s+} \subseteq \mathcal{F}_{t+}$.
    \item If $t$ is not isolated on the right, then $\mathcal{F}_{t+} = \bigsqcap_{u > t} \mathcal{F}_{u}$. Let $u > t$. Then $s \le u$, thus $\mathcal{F}_s \subseteq \mathcal{F}_u$. This proves that $\mathcal{F}_s \subseteq \bigsqcap_{u > t}, \mathcal{F}_u$, and thus $\mathcal{F}_{s+} \subseteq \mathcal{F}_{t+}$.
  \end{itemize}
  \item Suppose now that $s$ is not isolated on the right, so that $\mathcal{F}_{s+} = \bigsqcup_{u > s} \mathcal{F}_u$.
  \begin{itemize}
    \item If $t$ is isolated on the right, then $\mathcal{F}_{t+} = \mathcal{F}_t$. As $s$ is not isolated on the right, we deduce that $s \ne t$, and thus $s < t$ because $T$ is a partial order. Therefore $\bigsqcap_{u > s} \mathcal{F}_u \subseteq \mathcal{F}_t$, and thus $\mathcal{F}_{s+} \subseteq \mathcal{F}_{t+}$.
    \item If $t$ is not isolated on the right, then $\mathcal{F}_{t+} = \bigsqcap_{u > t} \mathcal{F}_u$. For any $u > t$, we have $u > s$ and thus $\bigsqcap_{v > s} \mathcal{F}_v \subseteq \mathcal{F}_u$, proving that $\bigsqcap_{u > s} \mathcal{F}_u \subseteq \bigsqcap_{u > t} \mathcal{F}_u$, and thus $\mathcal{F}_{s+} \subseteq \mathcal{F}_{t+}$.
  \end{itemize}
\end{itemize}

Turn now to the proof that for all $t \in T$, $\mathcal{F}_{t+} \subseteq \mathcal{A}$. Let $t \in T$. If $t$ is isolated on the right, then $\mathcal{F}_{t+} = \mathcal{F}_t \subseteq \mathcal{A}$ by definition of a filtration. Otherwise there exists $u > t$, and $\mathcal{F}_{t+} \subseteq \mathcal{F}_u \subseteq \mathcal{A}$, so we are done.
\end{proof}

\begin{lemma}\label{lem:leRightCont}
  \uses{def:rightLimitFiltration}
  \leanok
  \lean{MeasureTheory.Filtration.le_rightCont}
The filtration $\mathcal{F}$ is contained in its right continuation.
\end{lemma}

\begin{proof}\leanok
  \uses{lem:rightContDef}
Endow $T$ with the order topology, and consider $t \in T$. Using Lemma~\ref{lem:rightContDef}, we split into two cases. If $t$ is isolated on the right, then $\mathcal{F}_{t+} = \mathcal{F}_t \supseteq \mathcal{F}_t$ and we are done. Otherwise, for all $u > t$, $\mathcal{F}_t \subseteq \mathcal{F}_u$, therefore $\mathcal{F}_t \subseteq \bigsqcap_{u > t} \mathcal{F}_u$, and we are done.
\end{proof}


\begin{lemma}\label{lem:rightContSelf}
  \uses{def:rightLimitFiltration}
  \leanok
  \lean{MeasureTheory.Filtration.rightCont_self}
The right continuation of the right continuation of $\mathcal{F}$ is equal to the right continuation of $\mathcal{F}$.
\end{lemma}

\begin{proof}\leanok
  \uses{lem:leRightCont, lem:rightContDef}
Let $t \in T$. From Lemma~\ref{lem:leRightCont}, we already now that $\mathcal{F}_{t+} \subseteq \mathcal{F}_{t++}$. Endow $T$ with the order topology and split according to Lemma~\ref{lem:rightContDef}. If $t$ is isolated on the right, then $\mathcal{F}_{t++} = \mathcal{F}_{t+}$ and we are done. \end{proof}

\begin{definition}[Predictable $\sigma$-algebra]\label{def:predictableMeasurableSpace}
  \uses{def:filtration}
  \mathlibok
  \lean{MeasureTheory.Filtration.predictable}
Let $\mathcal{F}$ be a filtration on a measurable space indexed $\Omega$ by a linearly ordered set $T$.
Let $S = \{\{\bot\} \times A \mid A \in \mathcal{F}_\bot\}$ if $T$ has a bottom element and $S = \emptyset$ otherwise.
The predictable sigma-algebra on $T \times \Omega$ is the sigma-algebra generated by the set of sets $\{(t, \infty] \times A \mid t \in T, \: A \in \mathcal{F}_t\} \cup S$.
\end{definition}


\begin{definition}[Predictable process]\label{def:predictable}
  \uses{def:predictableMeasurableSpace}
  \mathlibok
  \lean{MeasureTheory.IsStronglyPredictable}
A process $X : T \to \Omega \to E$ is said to be predictable with respect to a filtration $\mathcal{F}$ if it is measurable with respect to the predictable sigma-algebra on $T \times \Omega$.
\end{definition}


\begin{lemma}\label{lem:Predictable.progressive}
  \uses{def:predictable, def:IsStronglyProgressive}
  \mathlibok
  \lean{MeasureTheory.IsStronglyPredictable.isStronglyProgressive}
A predictable process is progressively measurable.
\end{lemma}

\begin{proof}\leanok
Let $X : T \times \Omega \to E$ be a predictable process, we will show that it is progressively measurable. Namely, fixing $t \in T$, denoting
$$\iota_t : [0, t] \to T : s \mapsto s$$
we need to show that $\iota_t \circ X : [0, t] \times \Omega \to E$ is measurable with respect to $\mathcal{B}([0, t]) \otimes \mathcal{F}_t$.

Denoting $\Sigma_{\mathcal{F}}$ for the predictable $\sigma$-algebra generated by $\mathcal{F}$, as $u$ is predictable, we have that $X^{-1}(\mathcal{B}(E)) \le \Sigma_{\mathcal{F}}$. Thus, to show that $\iota_t \circ X$ is $\mathcal{B}([0, t]) \otimes \mathcal{F}_t$-measurable, it suffices to show that $\iota_t^{-1}(\Sigma_{\mathcal{F}}) \le \mathcal{B}([0, t]) \otimes \mathcal{F}_t$. In particular, as
$$\Sigma_{\mathcal{F}} = \sigma(\{(s, \infty) \times A \mid A \in \mathcal{F}_s\} \cup \{\{\perp\} \times A \mid A \in \mathcal{F}_\perp\})$$
is suffices to show that sets of the form $\iota_t^{-1}((s, \infty) \times A)$ for some $s \in T, A \in \mathcal{F}_s$ and $\iota_t^{-1}(\{\bot\} \times A)$ for some $A \in \mathcal{F}_\bot$ are $\mathcal{B}([0, t]) \otimes \mathcal{F}_t$-measurable.

Indeed, if $A \in \mathcal{F}_\bot$
$$\iota_t^{-1}(\{\bot\} \times A) = \{\bot\} \times A$$
while for any $s \in T$ and $A \in \mathcal{F}_s$,
$$\iota_t^{-1}((s, \infty) \times A) = \begin{cases}
    \varnothing, & t < s\\
    (s, t] \times A, & s \le t.
\end{cases}$$
By the monotonicity of the filtration $\mathcal{F}$, all of these cases are $\mathcal{B}([0, t]) \otimes \mathcal{F}_t$-measurable allowing us to conclude.
\end{proof}

\begin{lemma}\label{lem:predictable_nat_iff}
  \uses{def:predictable}
  \mathlibok
  \lean{MeasureTheory.IsStronglyPredictable.iff_measurable_add_one}
Let $X : \mathbb{N} \to \Omega \to E$ be a stochastic process and let $\mathcal{F}$ be a filtration indexed by $\mathbb{N}$.
Then $X$ is predictable if and only if $X_0$ is $\mathcal{F}_0$-measurable and for all $n \in \mathbb{N}$, $X_{n+1}$ is $\mathcal{F}_n$-measurable.
\end{lemma}


\begin{proof}\leanok
  \uses{lem:predictable_Ioc_prod}
Suppose first that $X$ is predictable. Straightaway, $X_0$ is $\mathcal{F}_0$-measurable as predictable implies progressively measurable which in turn implies adapted.

Fixing $n$, we observe that for any $S \in \mathcal{B}(E)$,
$$X_{n + 1}^{-1}(S) = \{\omega \mid (n + 1, \omega) \in X^{-1}(S)\} = \pi^{-1}(\iota^{-1}(X^{-1}(S)))$$
where
$$\pi : \Omega \to \{n + 1\} \times \Omega : \omega \mapsto (n + 1, \omega)$$
and
$$\iota : \{n + 1\} \times \Omega \to T \times \Omega : (n + 1, \omega) \mapsto (n + 1, \omega).$$
As $X^{-1}(S) \in \Sigma_{\mathcal{F}}$ -- the predictable $\sigma$-algebra, it suffices to show that $\pi^{-1}(\iota^{-1}(\Sigma_{\mathcal{F}})) \in \mathcal{F}_n$. To this end, we again only need to show these for the generating sets of $\Sigma_{\mathcal{F}}$:
\begin{itemize}
    \item For $A \in \mathcal{F}_0$, measurability is clear as $\iota^{-1}(\{0\} \times A) = \varnothing$.
    \item Similarly, for $m > n$ and $A \in \mathcal{F}_m$, $\iota^{-1}((m, \infty) \times A) = \varnothing$.
    \item For $m \le n$ and $A \in \mathcal{F}_m \le \mathcal{F}_n$ we have that $\pi^{-1}(\iota^{-1}((m, \infty) \times A)) = A$ which is $\mathcal{F}_n$ measurable by the monotonicity of the filtration.
\end{itemize}

Now, supposing $X_0$ is $\mathcal{F}_0$-measurable and $X_{n + 1}$ is $\mathcal{F}_n$-measurable, we will show that $X$ is predictable. Indeed, fixing $S \in \mathcal{B}(E)$, we have
$$X^{-1}(S) = \bigcup_{n \in \mathbb{N}} \{n\} \times X_n^{-1}(S) = {0} \times X_0^{-1}(S) \cup \bigcup_{n \in \mathbb{N}} \{n + 1\} \times X_{n + 1}^{-1}(S).$$
Thus, as $\{0\} \times X_0^{-1}(S) \in \Sigma_{\mathcal{F}}$ by construction and $\{n + 1\} \times X_{n + 1}^{-1}(S) = (n, n + 1] \times X_{n + 1}^{-1}(S) \in \Sigma_{\mathcal{F}}$ by Lemma~\ref{lem:predictable_Ioc_prod} and the fact that $X_{n + 1}^{-1}(S) \in \mathcal{F}_n$, we have that $X^{-1}(S) \in \Sigma_{\mathcal{F}}$ as required.
\end{proof}

\begin{lemma}\label{lem:condExpUI}
  \leanok
  \lean{MeasureTheory.UniformIntegrable.condExp'}
  If $(X_i)_{i \in \iota}$ is a family of (probabilistically) uniformly integrable functions and $(\mathcal{F}_j)_{j \in \kappa}$ is a family of $\sigma$-algebras,
  then the family $(P[X_i \mid \mathcal{F}_j])_{i \in \iota, j \in \kappa}$ is uniformly integrable.
\end{lemma}

\begin{proof}\leanok
  Since $(X_i)_{i \in \iota}$ is uniformly integrable, it is uniformly bounded in $L^1$, thus so is $(P[X_i \mid \mathcal{F}_j])_{i \in \iota, j \in \kappa}$. Moreover, for any \(\epsilon > 0\),
  there exists some \(\delta > 0\) such that
  for any measurable set \(A\) with \(P(A) < \delta\), we have that \(\sup_{i \in \iota} P[|X_i| \mathbb{I}_A] < \epsilon\).

  On the other hand, by Markov's inequality, for any $\lambda > 0$, $i \in \iota$ and $j \in \kappa$ we have that
  \[P(|P[X_i \mid \mathcal{F}_j]| \ge \lambda) \le \lambda^{-1}P[|P[X_i \mid \mathcal{F}_j]|] \le \lambda^{-1}P[|X_i|].\]
  Now set \(\lambda := \delta^{-1} \sup_{i \in \iota} P[|X_i|] + 1\). Then for any $i \in \iota$ and $j \in \kappa$ we have that
  \[P(|P[X_i \mid \mathcal{F}_j]| \ge \lambda) \le \frac{P[|X_i|]}{\delta^{-1} \sup_{k \in \iota} P[|X_k|] + 1} < \delta,\]
  and so,
  \begin{align*}
    P[|P[X_i \mid \mathcal{F}_j]| \mathbb{I}_{|P[X_i \mid \mathcal{F}_i]| \ge \lambda}]
    & = P[|P[X_i \mid \mathcal{F}_j] \mathbb{I}_{|P[X_i \mid \mathcal{F}_i]| \ge \lambda}|] \\
    & = P[|P[X_i \mathbb{I}_{|P[X_i \mid \mathcal{F}_i]| \ge \lambda} \mid \mathcal{F}_j]|] \\
    & \le P[P[|X_i|\mathbb{I}_{|P[X_i \mid \mathcal{F}_j]| \ge \lambda} \mid \mathcal{F}_j]] \\
    & = P[|X_i|\mathbb{I}_{|P[X_i \mid \mathcal{F}_j]| \ge \lambda}] < \epsilon,
  \end{align*}
  showing that \((P[X_i \mid \mathcal{F}_j])_{i \in \iota, j \in \kappa}\) is uniformly integrable.
\end{proof}


\begin{lemma}\label{lem:uniformIntegrable_of_bounded}
  \leanok
  \lean{MeasureTheory.uniformIntegrable_of_eLpNorm_le}
Let $(X_i)_{i \in \iota}$ be a family of random variables that is bounded in $\mathrm L^p$, for $p > q \ge 1$. Then $(X_i)_{i \in \iota}$ is uniformly integrable in $\mathrm L^q$.
\end{lemma}

\begin{proof}\leanok
  For any measurable set $s$ we have by Hölder inequality that
  $$\|X_t \mathbb{I}_s\|_q \le \|X_t\|_p P(s)^{1/q - 1/p},$$
  which goes to $0$ as $P(s)$ goes to zero. Moreover,
  $$\|X_t\|_q \le P(\Omega)^{1/q - 1/p} \|X_t\|_p,$$
  so $(X_i)_{i \in \iota}$ is bounded in $\mathrm L^q$.
\end{proof}

\begin{lemma}\label{lem:memLp_of_tendstoInMeasure}
  \leanok
  \lean{MeasureTheory.UniformIntegrable.memLp_of_tendstoInMeasure}

  in probability as $n \to \infty$. Then, $X$ is $L^p$.
\end{lemma}

\begin{proof}
  \leanok
  Since $X_n \to X$ in probability, it has a subsequence $(X_{n_k}) \subseteq (X_n)$ which converges
  to $X$ almost surely. Thus, we have by Fatou's lemma that
  \[P[|X|^p] = P[\liminf_{k \to \infty} |X_{n_k}|^p] \le \liminf_{k \to \infty} P[|X_{n_k}|^p] < \infty\]
  where the last inequality follows as uniform integrability implies that $(X_n)$ is uniform bounded in $L^p$.
\end{proof}

  \leanok

  Let $(t_n)_{n \in \mathbb{N}}$ be a sequence in $T$ such that $X_{t_n}$ converges almost everywhere to $Y$. We have by Fatou's lemma that

  This proves that the family of limits in a.e convergence of sequences of $X$ is $p$-uniformly integrable in the measure theory sense. One can prove uniform boundedness of this family by using Fatou's lemma in a similar way.
\end{proof}

\begin{definition}[Ordered Module]\label{def:isOrderedModule}
  \mathlibok
  \lean{IsOrderedModule}
Let $\alpha, \beta$ be preorders with $0$ elements and such that there is a scalar multiplication $(\_ \cdot \_) : \alpha \times \beta \to \beta$. Then $\beta$ is said to be an \emph{ordered $\alpha$-module} (or \emph{ordered module} if $\alpha$ is clear from the context) if the following hold:
\begin{itemize}
  \item $\forall a \in \alpha, \forall b_1, b_2 \in \beta, 0 \le a \implies b_1 \le b_2 \implies a \cdot b_1 \le a \cdot b_2$;
  \item $\forall a_1, a_2 \in \alpha, \forall b \in \beta, 0 \le b \implies a_1 \le a_2 \implies a_1 \cdot b \le a_2 \cdot b$.
\end{itemize}
\end{definition}

\begin{definition}[Order-closed topology]\label{def:orderClosedTopology}
  \mathlibok
  \lean{OrderClosedTopology}
Let $X$ be a topological space that is also a preorder. The space $X$ is set to be \emph{order-closed}, or to have \emph{order-closed topology}, if the set $\{(x, y) \in X \times X \mid x \le y\}$ is closed.
\end{definition}

\begin{lemma}\label{lem:optionalSampling_discrete_submartingale}
  \uses{def:orderClosedTopology, def:isOrderedAddMonoid, def:isOrderedModule, def:Submartingale, def:IsStoppingTime}
  \leanok
  \lean{MeasureTheory.Submartingale.stoppedValue_min_ae_le_condExp_nat}
Let $X$ be a discrete time submartingale with respect to the filtration $\mathcal{F}$ taking values in a real Banach space $E$. Assume $E$ is an order-closed partial order, an ordered monoid and an ordered module. Let $\tau, \sigma$ be stopping times. Then, if $\tau$ is bounded, we have that almost surely, $X_{\tau \wedge \sigma} \le P[X_{\tau} \mid \mathcal{F}_{\sigma}]$.
\end{lemma}

\begin{proof}
  \uses{lem:martingale_martingalePart, lem:predictable_predictablePart, lem:optionalSampling_discrete, lem:Submartingale.monotone_predictablePart, lem:centering_basic}
Use Doob decomposition to write $X_n = M_n + A_n$, where $M$ (Definition~\ref{def:martingalePart}) is a martingale (Lemma~\ref{lem:martingale_martingalePart}) and $A$ (Definition~\ref{def:predictablePart}) is a predictable process (Lemma~\ref{lem:predictable_predictablePart}). By Lemma~\ref{lem:optionalSampling_discrete}, we have that almost surely, $M_{\tau \wedge \sigma} = P[M_{\tau} \mid \mathcal{F}_{\sigma}]$. Because $A$ is predictable and $\tau \wedge \sigma \le \sigma$, we deduce that almost surely, $A_{\tau \wedge \sigma} = P[A_{\tau \wedge \sigma} \mid \mathcal{F}_\sigma]$. Moreover, by Lemma~\ref{lem:Submartingale.monotone_predictablePart}, we know that almost surely, $A$ is nondecreasing. Therefore, using the fact $\tau \wedge \sigma \le \tau$, we get that $P[A_{\tau \wedge \sigma} \mid \mathcal{F}_\sigma] \le P[A_\tau \mid \mathcal{F}_\sigma]$. We deduce that almost surely,
$$X_{\tau \wedge \sigma} = M_{\tau \wedge \sigma} + A_{\tau \wedge \sigma} \le P[M_\tau \mid \mathcal{F}_\sigma] + P[A_\tau \mid \mathcal{F}_\sigma] = P[X_\tau \mid \mathcal{F}_\sigma],$$
concluding the proof.
\end{proof}

\begin{lemma}[Optional sampling (continuous time)]\label{lem:optionalSampling}
  \uses{def:Martingale, def:IsStoppingTime, def:RightContinuous, def:approximableTimeIndex}
  \leanok
  \lean{MeasureTheory.Martingale.stoppedValue_min_ae_eq_condExp'}
  Let $X$ be a right-continuous $\mathcal{F}$-martingale on an approximable time index. Then, for
  any stopping times $\sigma, \tau$ with $\tau$ bounded, we have that
  $X_{\sigma \wedge \tau} = P[X_{\tau} \mid \mathcal{F}_{\sigma}]$ almost surely.
\end{lemma}

\begin{proof}
  \leanok
  \uses{lem:StoppingTimeGenMono, lem:uniformIntegrable_stoppedValue_martingale,
    lem:tendsto_eLpNorm_stoppedValue_discreteApproxSequence}
  Fixing $A \in \mathcal{F}_{\sigma}$, we need to show that $P[X_{\tau} \mathbb{I}_A] = P[X_{\sigma \wedge \tau} \mathbb{I}_A]$.

  Let $(\tau_n), (\sigma_n)$ be discrete approximation sequences of $\tau$ and $\sigma$ respectively.
  As $\tau_n, \sigma_n$ take values in a countable set,
  we have by the discrete time optional sampling theorem (Lemma~\ref{lem:optionalSampling_discrete}) that
  $$X_{\sigma_n \wedge \tau_n} = P[X_{\tau_n} \mid \mathcal{F}_{\sigma_n}]$$
  and so, as \(\mathcal{F}_{\sigma} \subseteq \mathcal{F}_{\sigma_n}\) by Lemma~\ref{lem:StoppingTimeGenMono},
  we have that $P[X_{\sigma_n \wedge \tau_n} \mathbb{I}_A] = P[X_{\tau_n} \mathbb{I}_A]$.
  On the other hand, by Lemma~\ref{lem:uniformIntegrable_stoppedValue_martingale}, the families $\{X_{\tau_n}\}$ and $\{X_{\sigma_n \wedge \tau_n}\}$ are uniformly integrable.
  Thus, as $X$ is right-continuous, $(X_{\sigma_n \wedge \tau_n}, X_{\tau_n}) \to (X_{\sigma \wedge \tau}, X_{\tau})$ a.s.
  We have $P[X_{\tau} \mathbb{I}_A] = P[X_{\sigma \wedge \tau} \mathbb{I}_A]$
  by Lemma~\ref{lem:tendsto_eLpNorm_stoppedValue_discreteApproxSequence} as desired.
\end{proof}

\begin{lemma}[Optional sampling for real submartingales]\label{lem:Submartingale.setIntegral_le_setIntegral_stoppedValue}
  \uses{def:Submartingale, def:IsStoppingTime, def:RightContinuous, def:approximableTimeIndex}
  \leanok
  \lean{MeasureTheory.Submartingale.setIntegral_le_setIntegral_stoppedValue}
  Let $X$ be a right-continuous real $\mathcal{F}$-submartingale on an approximable time index, for a finite measure $P$.
  Let $\tau$ be a stopping time bounded by $t$, and let $C \in \mathcal{F}_s$ be such that $s \le \tau$ on $C$.
  Then $P[X_s \mathbb{I}_C] \le P[X_\tau \mathbb{I}_C]$.
\end{lemma}

\begin{proof}
  \leanok
  \uses{def:approxSeq, lem:tendsto_stoppedValue_discreteApproxSequence, lem:uniformIntegrable_stoppedValue_martingale_of_countable_range, lem:uniformIntegrableDominated, lem:vitali}
  \emph{Finite range.} If $\tau$ takes values in a finite set, we prove the inequality by induction on the largest value $a$ of $\tau$.
  If no other value of $\tau$ is at least $s$, then $\tau = a$ on $C$ and the inequality is the submartingale property.
  Otherwise let $u \ge s$ be the second largest value. Then $\tau \wedge u$ takes fewer values, $P[X_s \mathbb{I}_C] \le P[X_{\tau \wedge u} \mathbb{I}_C]$ by induction, and
  $P[X_{\tau \wedge u} \mathbb{I}_C] \le P[X_\tau \mathbb{I}_C]$ since the two integrands differ only on $C \cap \{\tau > u\} \in \mathcal{F}_u$, on which they are equal to $X_u$ and $X_a$.

  \emph{Countable range.} If $\tau$ takes countably many values $(t_i)_{i \in \mathbb{N}}$, let $\rho_m$ be equal to $\tau$ if $\tau \in \{t_0, \ldots, t_{m-1}\}$ and to $t$ otherwise.
  This is a stopping time with finitely many values, with $s \le \rho_m$ on $C$, and $X_{\rho_m}$ is eventually equal to $X_\tau$ and dominated by $\vert X_\tau \vert + \vert X_t \vert$, which is integrable.
  We conclude by dominated convergence.

  \emph{$X$ bounded from below.} Suppose that $X \ge c$ and let $(\tau_n)$ be a discrete approximation sequence of $\tau$, bounded by $t$.
  By the previous case, $P[X_s \mathbb{I}_C] \le P[X_{\tau_n} \mathbb{I}_C]$ for all $n$, and $X_{\tau_n} \to X_\tau$ almost surely by right-continuity (Lemma~\ref{lem:tendsto_stoppedValue_discreteApproxSequence}).
  For $i \le t$ we have $\vert X_i \vert \le N_i := P[\vert X_t \vert + \vert c \vert \mid \mathcal{F}_i]$, hence $\vert X_{\tau_n} \vert \le N_{\tau_n}$ almost surely.
  Since $N$ is a martingale, $(N_{\tau_n})$ is uniformly integrable (Lemma~\ref{lem:uniformIntegrable_stoppedValue_martingale_of_countable_range}), hence so is $(X_{\tau_n})$ (Lemma~\ref{lem:uniformIntegrableDominated}).
  By the Vitali convergence theorem (Lemma~\ref{lem:vitali}), $X_{\tau_n} \to X_\tau$ in $L^1$, which gives the inequality.

  \emph{General case.} For $m \in \mathbb{N}$, the process $X \vee (-m)$ is a right-continuous submartingale bounded from below, hence
  $P[(X_s \vee (-m)) \mathbb{I}_C] \le P[(X_\tau \vee (-m)) \mathbb{I}_C]$.
  Since $\vert x \vee (-m) \vert \le \vert x \vert$ and $X_\tau$ is integrable, we can let $m \to \infty$ by dominated convergence.

  The truncation avoids proving that $(X_{\tau_n})$ is uniformly integrable for a general submartingale, which would require a backward submartingale argument.
\end{proof}

\begin{lemma}[Optional stopping for real submartingales]\label{lem:Submartingale.stoppedProcess}
  \uses{def:Submartingale, def:IsStoppingTime, def:RightContinuous, def:approximableTimeIndex, def:stoppedProcess}
  \leanok
  \lean{MeasureTheory.Submartingale.stoppedProcess_of_approximable}
  Let $X$ be a right-continuous real $\mathcal{F}$-submartingale on an approximable time index, for a finite measure $P$, and let $\tau$ be a stopping time.
  Then the stopped process $X^\tau$ is a submartingale.
\end{lemma}

\begin{proof}
  \leanok
  \uses{lem:Submartingale.setIntegral_le_setIntegral_stoppedValue}
  The stopped process is adapted, and it is integrable since $X_{\tau \wedge t}$ is integrable for all $t$.
  Let $s \le t$ and $B \in \mathcal{F}_s$. We have to prove that $P[X_{\tau \wedge s} \mathbb{I}_B] \le P[X_{\tau \wedge t} \mathbb{I}_B]$.
  On $B \cap \{\tau \le s\}$, both integrands are equal to $X_\tau$.
  The set $C = B \cap \{s < \tau\}$ belongs to $\mathcal{F}_s$, $X_{\tau \wedge s} = X_s$ on $C$, and $s \le \tau \wedge t$ on $C$,
  hence $P[X_s \mathbb{I}_C] \le P[X_{\tau \wedge t} \mathbb{I}_C]$ by Lemma~\ref{lem:Submartingale.setIntegral_le_setIntegral_stoppedValue} applied to the stopping time $\tau \wedge t$.
\end{proof}

\begin{lemma}\label{lem:Submartingale.setIntegral_stoppedValue_le_setIntegral}
  \uses{def:Submartingale, def:IsStoppingTime, def:RightContinuous, def:approximableTimeIndex}
  \leanok
  \lean{MeasureTheory.Submartingale.setIntegral_stoppedValue_le_setIntegral}
  Let $X$ be a right-continuous real $\mathcal{F}$-submartingale on an approximable time index, for a finite measure $P$.
  Let $\tau$ be a stopping time bounded by $t$ and let $C \in \mathcal{F}_\tau$.
  Then $P[X_\tau \mathbb{I}_C] \le P[X_t \mathbb{I}_C]$.
\end{lemma}

\begin{proof}
  \leanok
  \uses{def:approxSeq, lem:StoppingTimeGenMono, lem:Submartingale.setIntegral_le_setIntegral_stoppedValue}
  \emph{Countable range.} If $\tau$ takes countably many values, then for each value $i$ the set $C \cap \{\tau = i\}$ belongs to $\mathcal{F}_i$, hence
  $P[X_\tau \mathbb{I}_{C \cap \{\tau = i\}}] = P[X_i \mathbb{I}_{C \cap \{\tau = i\}}] \le P[X_t \mathbb{I}_{C \cap \{\tau = i\}}]$ by the submartingale property.
  We sum over the values of $\tau$, which is possible since $X_\tau$ is integrable.

  \emph{$X$ bounded from below.} Let $(\tau_n)$ be a discrete approximation sequence of $\tau$, bounded by $t$.
  Since $\mathcal{F}_\tau \subseteq \mathcal{F}_{\tau_n}$ (Lemma~\ref{lem:StoppingTimeGenMono}), the previous case gives $P[X_{\tau_n} \mathbb{I}_C] \le P[X_t \mathbb{I}_C]$ for all $n$,
  and $X_{\tau_n} \to X_\tau$ in $L^1$ as in the proof of Lemma~\ref{lem:Submartingale.setIntegral_le_setIntegral_stoppedValue}.

  \emph{General case.} We apply the previous case to $X \vee (-m)$ and let $m \to \infty$ by dominated convergence, as in the proof of Lemma~\ref{lem:Submartingale.setIntegral_le_setIntegral_stoppedValue}.
\end{proof}

\begin{lemma}[Optional sampling theorem for real submartingales]\label{lem:optionalSamplingSubmartingale}
  \uses{def:Submartingale, def:IsStoppingTime, def:RightContinuous, def:approximableTimeIndex}
  \leanok
  \lean{MeasureTheory.Submartingale.stoppedValue_min_ae_le_condExp}
  Let $X$ be a right-continuous real $\mathcal{F}$-submartingale on an approximable time index, for a finite measure $P$. Then, for
  any stopping times $\sigma, \tau$ with $\tau$ bounded, we have that
  $X_{\sigma \wedge \tau} \le P[X_{\tau} \mid \mathcal{F}_{\sigma}]$ almost surely.
\end{lemma}

\begin{proof}
  \leanok
  \uses{lem:Submartingale.stoppedProcess, lem:Submartingale.setIntegral_stoppedValue_le_setIntegral}
  Let $k$ be a bound of $\tau$ and let $A \in \mathcal{F}_{\sigma}$: we need to show that $P[X_{\sigma \wedge \tau} \mathbb{I}_A] \le P[X_{\tau} \mathbb{I}_A]$.

  The stopped process $Y = X^\tau$ is a right-continuous submartingale (Lemma~\ref{lem:Submartingale.stoppedProcess}), with $Y_k = X_\tau$ and $Y_{\sigma \wedge k} = X_{\sigma \wedge \tau}$.
  On $A \cap \{\sigma > k\}$ we have $Y_{\sigma \wedge k} = Y_k$.
  The set $A \cap \{\sigma \le k\}$ belongs to $\mathcal{F}_{\sigma \wedge k}$, hence
  $P[Y_{\sigma \wedge k} \mathbb{I}_{A \cap \{\sigma \le k\}}] \le P[Y_k \mathbb{I}_{A \cap \{\sigma \le k\}}]$
  by Lemma~\ref{lem:Submartingale.setIntegral_stoppedValue_le_setIntegral} applied to $Y$ and to the stopping time $\sigma \wedge k$.
\end{proof}

\begin{lemma}[Optional sampling theorem for nonnegative submartingales in a Banach lattice]\label{lem:optionalSamplingSubmartingale_of_nonneg}
  \uses{def:Submartingale, def:IsStoppingTime, def:RightContinuous, def:approximableTimeIndex}
  \leanok
  \lean{MeasureTheory.Submartingale.stoppedValue_min_ae_le_condExp_of_nonneg}
  Let $X$ be a nonnegative right-continuous $\mathcal{F}$-submartingale with values in a separable Banach lattice, on an approximable time index, for a finite measure $P$. Then, for
  any stopping times $\sigma, \tau$ with $\tau$ bounded, we have that
  $X_{\sigma \wedge \tau} \le P[X_{\tau} \mid \mathcal{F}_{\sigma}]$ almost surely.
\end{lemma}

\begin{proof}
  \uses{lem:optionalSamplingSubmartingale}
  For a positive continuous linear form $\varphi$, the process $\varphi \circ X$ is a nonnegative right-continuous real submartingale, to which we can apply Lemma~\ref{lem:optionalSamplingSubmartingale}.
  Countably many positive continuous linear forms determine the order of a separable Banach lattice.

  Let $k$ be a bound of $\tau$. Since $X_k$ is integrable, $\varphi(P[X_k \mid \mathcal{F}_\tau]) = P[\varphi(X_k) \mid \mathcal{F}_\tau] \ge \varphi(X_\tau)$ for all positive $\varphi$, hence $0 \le X_\tau \le P[X_k \mid \mathcal{F}_\tau]$ almost surely and $X_\tau$ is integrable.
  Then $\varphi(P[X_\tau \mid \mathcal{F}_\sigma]) = P[\varphi(X_\tau) \mid \mathcal{F}_\sigma] \ge \varphi(X_{\sigma \wedge \tau})$ for all positive $\varphi$, which gives the result.

  The nonnegativity assumption cannot be removed: without it, $X_\tau$ is not integrable in general, even for a stopping time with countable range and $E = \ell^2$ with the coordinatewise order.
\end{proof}

\section{Martingale convergence}

\begin{definition}\label{def:limitProcess}
  \uses{def:filtration}
  \mathlibok
  \lean{MeasureTheory.Filtration.limitProcess}
Let $X : T \to \Omega \to E$ be a stochastic process, let $\mathcal{F}$ be a filtration on $\Omega$ indexed by $T$ and let $P$ be a measure on $\Omega$.
If there exists a function $Y : \Omega \to E$ which is measurable with respect to $\mathcal{F}_\infty$ such that for $P$-almost surely, $X_t$ converges to $Y$ as $t$ goes to infinity, then we say that $Y$ is the limit of $X$.
We denote it by $X_\infty$.
\end{definition}


In Mathlib, we have results about convergence of martingales to their limit in discrete time.


For martingales with values in a Banach space $E$, the convergence of uniformly integrable martingales requires an assumption on $E$.

\begin{definition}[Radon-Nikodym property]\label{def:HasRadonNikodymProp}
  \leanok
  \lean{HasRadonNikodymProp}
A Banach space $E$ has the Radon-Nikodym property if for every finite measure $\mu$, every $E$-valued vector measure of bounded variation which is absolutely continuous with respect to $\mu$ has a Bochner integrable density with respect to $\mu$.
\end{definition}

\begin{lemma}\label{lem:HasRadonNikodymProp_stability}
  \uses{def:HasRadonNikodymProp}
  \leanok
  \lean{HasRadonNikodymProp.completeSpace, HasRadonNikodymProp.of_antilipschitzWith, HasRadonNikodymProp.of_continuousLinearEquiv, HasRadonNikodymProp.submodule_of_isClosed, HasRadonNikodymProp.prod, HasRadonNikodymProp.pi, HasRadonNikodymProp.of_finiteDimensional}
A normed space with the Radon-Nikodym property is complete.
If $E$ has the Radon-Nikodym property, $F$ is a Banach space and $L : F \to E$ is a continuous linear map with $\Vert x \Vert \le c \Vert L x \Vert$ for all $x$, then $F$ has the Radon-Nikodym property.
In particular the Radon-Nikodym property is invariant under linear isomorphisms and is inherited by closed subspaces.
The space $\mathbb{R}$, finite products of spaces with the Radon-Nikodym property and finite-dimensional spaces have the Radon-Nikodym property.
\end{lemma}

\begin{proof}
  \leanok
For the completeness, apply the property to the measure $x \delta$ on a one-point space: if $E$ was not complete, its density would have Bochner integral $0$, hence $x = 0$ for all $x \in E$.

Let $\nu$ be an $F$-valued measure of bounded variation, absolutely continuous with respect to $\mu$.
Then $L \circ \nu$ is an $E$-valued measure with the same properties: let $f$ be its density.
The image of $f$ in the quotient of $E$ by the closed subspace $L(F)$ has integral zero on all measurable sets, hence $f \in L(F)$ almost everywhere.
Since $L$ is an embedding, $g = L^{-1} \circ f$ is measurable, it is integrable since $\Vert g \Vert \le c \Vert f \Vert$, and it is a density of $\nu$ since $L$ is injective.

The case of $\mathbb{R}$ is the Radon-Nikodym theorem for signed measures.
For a product, the densities of the coordinates of $\nu$ are the coordinates of a density of $\nu$.
A finite-dimensional space is isomorphic to a finite product of copies of $\mathbb{R}$.
\end{proof}

\begin{lemma}\label{lem:HasRadonNikodymProp_of_forall_isSeparable}
  \uses{def:HasRadonNikodymProp}
  \leanok
  \lean{HasRadonNikodymProp.of_forall_isSeparable}
The Radon-Nikodym property is separably determined: a normed space has the Radon-Nikodym property if all its closed separable subspaces have it.
\end{lemma}

\begin{proof}
  \leanok
The space $E$ is complete: a Cauchy sequence takes its values in a closed separable subspace, which has the Radon-Nikodym property and is thus complete.
Let $\nu$ be an $E$-valued measure with $\Vert \nu(A) \Vert \le \mu(A)$ for all measurable sets $A$ (the general case reduces to this one by taking for $\mu$ the variation of $\nu$).
On a countably generated sub-$\sigma$-algebra $m$, the range of $\nu$ is separable, hence contained in a closed separable subspace, and $\nu$ has a density $d_m$ on $m$, with norm at most $1$.
By L\'evy's upward theorem in $L^1$, the net $(d_m)$ is Cauchy in $L^1$, hence it is eventually constant, and its eventual value is a density of $\nu$.
\end{proof}

\begin{theorem}\label{thm:HasRadonNikodymProp_of_reflexive}
  \uses{def:HasRadonNikodymProp}
  \leanok
  \lean{HasRadonNikodymProp.of_isReflexive}
Reflexive Banach spaces have the Radon-Nikodym property. In particular real and complex Hilbert spaces have the Radon-Nikodym property.
\end{theorem}

\begin{proof}
  \leanok
  \uses{lem:HasRadonNikodymProp_of_forall_isSeparable}
A closed subspace of a reflexive space is reflexive, hence by Lemma~\ref{lem:HasRadonNikodymProp_of_forall_isSeparable} we can suppose that $E$ is separable.
Let $\nu$ be an $E$-valued measure with $\Vert \nu(A) \Vert \le \mu(A)$ for all measurable sets $A$ and let $(\varphi_k)$ be a sequence of continuous linear forms which separates the points of $E$.
Each real measure $\varphi_k \circ \nu$ has a density $g_k$ with respect to $\mu$.
For almost every $\omega$, we have $\vert \sum_k c_k g_k(\omega) \vert \le \Vert \sum_k c_k \varphi_k \Vert$ for all finitely supported families of coefficients $c$ (it suffices to check this for rational coefficients).
By the Hahn-Banach theorem the linear form $\sum_k c_k \varphi_k \mapsto \sum_k c_k g_k(\omega)$ extends to an element of the bidual of $E$ with norm at most $1$, hence by reflexivity it is the evaluation at a point $f(\omega)$ with $\Vert f(\omega) \Vert \le 1$.
The function $f$ is measurable since the $\varphi_k$ separate the points of the Polish space $E$, and it is a density of $\nu$.
\end{proof}

\begin{theorem}\label{thm:Martingale.exists_ae_eq_condExp_of_uniformIntegrable}
  \uses{def:Martingale, def:HasRadonNikodymProp}
  \leanok
  \lean{MeasureTheory.Martingale.exists_ae_eq_condExp_of_uniformIntegrable}
Let $X$ be a uniformly integrable martingale with values in a space with the Radon-Nikodym property, for a finite measure $P$.
Then $X$ is closed: there exists an integrable function $Z$, measurable with respect to $\mathcal{F}_\infty$, such that $X_t = P[Z \mid \mathcal{F}_t]$ almost surely for all $t$.
\end{theorem}

\begin{proof}
  \leanok
We can suppose that the $\sigma$-algebra is $\mathcal{F}_\infty$. The sets which are measurable at some time are then dense for the distance $P(A \triangle B)$, and for such a set $A \in \mathcal{F}_s$ the integral $P[X_t \mathbb{I}_A]$ does not depend on $t \ge s$.
By uniform integrability, $P[X_t \mathbb{I}_A]$ converges for every measurable set $A$ as $t \to \infty$, to a limit $\nu(A)$.
Again by uniform integrability, $\nu$ is a vector measure of bounded variation which is absolutely continuous with respect to $P$.
Its density $Z$ satisfies $P[Z \mathbb{I}_A] = P[X_s \mathbb{I}_A]$ for $A \in \mathcal{F}_s$.
\end{proof}

\begin{theorem}\label{thm:Martingale.ae_tendsto_of_ae_eq_condExp}
  \uses{def:Martingale, def:RightContinuous}
  \leanok
  \lean{MeasureTheory.Martingale.ae_tendsto_of_ae_eq_condExp}
Let $X$ be a right-continuous martingale with values in a Banach space, such that $X_t = P[Z \mid \mathcal{F}_t]$ almost surely for all $t$, for an integrable function $Z$ which is almost everywhere strongly measurable with respect to $\mathcal{F}_\infty$.
Then $X_t$ converges to $Z$ almost surely as $t$ goes to infinity.
\end{theorem}

\begin{proof}
  \leanok
  \uses{cor:doob_ineq_norm}
Let $(t_n)$ be a nondecreasing sequence of times tending to infinity. By L\'evy's upward theorem in $L^1$, $X_{t_n}$ tends to $Z$ in $L^1$.
For a time $s$, the process $(X_t - X_{t \wedge s})_t$ is a right-continuous martingale, and Doob's maximal inequality (Lemma~\ref{cor:doob_ineq_norm}) gives for $\varepsilon > 0$
$$P\left(\sup_{t \ge s} \Vert X_t - X_s \Vert \ge \varepsilon\right) \le \frac{1}{\varepsilon} \sup_{t \ge s} P[\Vert X_t - X_s \Vert] \le \frac{1}{\varepsilon} P[\Vert Z - X_s \Vert] \: .$$
We choose times $s_k$ in the sequence such that $P[\Vert Z - X_{s_k} \Vert] \le 4^{-k}$.
Then the events $\{\sup_{t \ge s_k} \Vert X_t - X_{s_k} \Vert \ge 2^{-k}\}$ and $\{\Vert Z - X_{s_k} \Vert \ge 2^{-k}\}$ have probability at most $2^{-k}$.
By the Borel-Cantelli lemma, almost surely $\Vert X_t - Z \Vert < 2 \cdot 2^{-k}$ for all $t \ge s_k$, for $k$ large enough.
\end{proof}

\begin{theorem}\label{thm:tendsto_limitProcess_of_cadlag}
  \uses{def:Martingale, def:limitProcess, def:IsCadlag, def:HasRadonNikodymProp}
  \leanok
  \lean{MeasureTheory.Martingale.ae_tendsto_limitProcess}
Let $X$ be an uniformly integrable cadlag martingale with respect to the filtration $\mathcal{F}$, with values in a space with the Radon-Nikodym property, for a finite measure $P$.
Then there exists a limit process $X_\infty$ measurable with respect to $\mathcal{F}_\infty$ such that $X_t$ converges to $X_\infty$ almost surely as $t$ goes to infinity.
Furthermore, $X_t = P[X_\infty \mid \mathcal{F}_t]$ almost surely.
\end{theorem}

\begin{proof}
  \leanok
  \uses{thm:Martingale.exists_ae_eq_condExp_of_uniformIntegrable, thm:Martingale.ae_tendsto_of_ae_eq_condExp}
By Theorem~\ref{thm:Martingale.exists_ae_eq_condExp_of_uniformIntegrable}, $X$ is closed by a function $Z$, and by Theorem~\ref{thm:Martingale.ae_tendsto_of_ae_eq_condExp} it converges almost surely to $Z$.
\end{proof}

\begin{lemma}\label{lem:limitProcess_ae_eq}
  \uses{def:limitProcess}
  \leanok
  \lean{MeasureTheory.Filtration.limitProcess_ae_eq}
Let $X : T \to \Omega \to E$ be a process such that there exists $g : \Omega \to E$ that is a.e. strongly measurable with respect to $\bigsqcup t, \mathcal{F}_t$ and such that $X$ converges almost surely to $g$. Then $X_\infty$ and $g$ are a.e. equal.
\end{lemma}

\begin{proof}
  Because there exist a function that is a.e. strongly measurable with respect to $\bigsqcup t, \mathcal{F}_t$ and towards which $X$ converges almost surely, $X_\infty$ is such a function by definition. Because $X$ converges almost surely towards $g$, $g$ and $X_\infty$ are equal on the set where $X$ converges to $X_\infty$ and $X$ converges to $g$ by uniqueness of the limit.
\end{proof}

\begin{lemma}\label{lem:limitProcess_congr}
  \uses{def:limitProcess}
  \leanok
  \lean{MeasureTheory.Filtration.limitProcess_congr}
  Let $X$ and $Y$ be two indistinguishable processes. Then $X_\infty$ and $Y_\infty$ are a.e. equal.
\end{lemma}

\begin{proof}
  \uses{lem:limitProcess_ae_eq}
  If there exists $g : \Omega \to E$ that is a.e. strongly measurable with respect to $\bigsqcup t, \mathcal{F}_t$ and such that $X$ converges almost surely to $g$, then it is enough to show that $Y$ converges almost surely to $g$ by Lemma~\ref{lem:limitProcess_ae_eq}. This follows from indistinguishability.

  Otherwise, $X_\infty = 0$, and by indistinguishability there does not exist $g : \Omega \to E$ that is a.e. strongly measurable with respect to $\bigsqcup t, \mathcal{F}_t$ and such that $Y$ converges almost surely to $g$, so $Y_\infty = 0$ as well.
\end{proof}

\begin{lemma}\label{lem:limitProcess_const}
  \uses{def:limitProcess}
  \leanok
  \lean{MeasureTheory.Filtration.limitProcess_congr}
  Let $c \in E$ and $X$ be the constant process equal to $c$. Then $X_\infty$ is a.e. equal to $c$.
\end{lemma}

\begin{proof}
  \uses{lem:limitProcess_ae_eq}
  This follows from Lemma~\ref{lem:limitProcess_ae_eq} as a constant function converges to a constant, and a constant function is a.e. strongly measurable with respect to any sigma-algebra.
\end{proof}

\begin{lemma}\label{lem:limitProcess_smul}
  \uses{def:limitProcess}
  \leanok
  \lean{MeasureTheory.Filtration.limitProcess_smul}
  If $X$ is a stochastic process and $c \in \mathbb{R}$, then $(c \cdot X)_\infty$ is a.e. equal to $c \cdot X_\infty$.
\end{lemma}

\begin{proof}
  \uses{lem:limitProcess_ae_eq, lem:limitProcess_const}
  If $c = 0$ we can conclude by Lemma~\ref{lem:limitProcess_const}. Otherwise, if there exists $g : \Omega \to E$ that is a.e. strongly measurable with respect to $\bigsqcup t, \mathcal{F}_t$ and such that $X$ converges almost surely to $g$, then $c \cdot X$ converges almost surely to $c \cdot g$ and we can conclude by Lemma~\ref{lem:limitProcess_ae_eq}. Otherwise there does not exist $g : \Omega \to E$ that is a.e. strongly measurable with respect to $\bigsqcup t, \mathcal{F}_t$ and such that $c \cdot X$ converges almost surely to $g$, because if it did it would also be the case for $X$, so $X_\infty$ and $(c \cdot X)_\infty$ are both a.e. equal to $0$.
\end{proof}

\begin{lemma}\label{lem:limitProcess_neg}
  \uses{def:limitProcess}

Otherwise there does not exist $g : \Omega \to E$ that is a.e. strongly measurable with respect to $\bigsqcup t, \mathcal{F}_t$ and such that $-X$ converges almost surely to $g$, because if it did it would also be the case for $X$, so $X_\infty$ and $(-X)_\infty$ are both a.e. equal to $0$.

\begin{theorem}\label{thm:condExp_limitProcess_stopped}
  \uses{def:limitProcess, def:Martingale}
  \leanok
  \lean{MeasureTheory.Martingale.condExp_limitProcess_ae_eq'}
  If $X$ is a càdlàg and uniformly integrable martingale with values in a space with the Radon-Nikodym property, on an approximable time index, then for any stopping time $\tau$, almost surely,
  $$P[X_\infty | \mathcal{F}_\tau] = X_\tau.$$
\end{theorem}

\begin{proof}
  \leanok
  \uses{thm:tendsto_limitProcess_of_cadlag, thm:condExp_limitProcess, lem:optionalSampling}
Let $(t_n)$ be a nondecreasing sequence of times tending to infinity.
By optional sampling for the bounded stopping time $\tau \wedge t_n$ (Lemma~\ref{lem:optionalSampling}) and Theorem~\ref{thm:condExp_limitProcess},
$X_{\tau \wedge t_n} = P[X_{t_n} \mid \mathcal{F}_{\tau \wedge t_n}] = P[X_\infty \mid \mathcal{F}_{\tau \wedge t_n}]$.
In particular $P[\Vert X_{\tau \wedge t_n} \Vert] \le P[\Vert X_\infty \Vert]$, and since $X_{\tau \wedge t_n}$ tends to $X_\tau$ almost surely (by Theorem~\ref{thm:tendsto_limitProcess_of_cadlag} on the event $\{\tau = \infty\}$), $X_\tau$ is integrable by Fatou's lemma.

Let $A \in \mathcal{F}_\tau$. The set $A \cap \{\tau \le t_n\}$ belongs to $\mathcal{F}_{\tau \wedge t_n}$ and $X_{\tau \wedge t_n} = X_\tau$ on that set, hence
$P[X_\tau \mathbb{I}_{A \cap \{\tau \le t_n\}}] = P[X_\infty \mathbb{I}_{A \cap \{\tau \le t_n\}}]$.
Letting $n$ tend to infinity, we get $P[X_\tau \mathbb{I}_{A \cap \{\tau < \infty\}}] = P[X_\infty \mathbb{I}_{A \cap \{\tau < \infty\}}]$, and $X_\tau = X_\infty$ on $\{\tau = \infty\}$.
\end{proof}

\begin{lemma}[Doob's maximal Inequality for countable]\label{lem:doob_countable}
  \uses{def:Submartingale}
  \leanok
  \lean{MeasureTheory.Submartingale.ae_bddAbove_Iic, ProbabilityTheory.maximal_ineq_countable}
  Let $X : I \rightarrow \Omega \rightarrow \mathbb{R}$ be a non-negative sub-martingale with $I$ countable.
  Then for every $M \in I,\lambda > 0$ and $p>1$ we have
  \begin{align*}
    P\left( \sup_{i\in I, i\leq M}X_i\geq\lambda \right)
    \le \frac{\mathbb{E}\left[X_M \mathbb{I}_{\sup_{i \le M}X_i \ge \lambda}\right]}{\lambda}
    \le \frac{\mathbb{E}[X_M]}{\lambda}
    \: .
  \end{align*}
\end{lemma}

\begin{proof}
  \uses{lem:maximal_ineq}
  \leanok
For any finite subset $J \subset I$ with $M \in J$, we have by Lemma~\ref{lem:maximal_ineq}
\begin{align*}
  P\left( \sup_{i\in J, i \le M}X_i\geq\lambda \right)
  \le \frac{\mathbb{E}\left[X_{M} \mathbb{I}_{\sup_{i \in J, i \le M}X_i \ge \lambda}\right]}{\lambda}
  \: .
\end{align*}
Then we build a countable increasing sequence of finite sets $J_n$ with $\bigcup_nJ_n=I$.
Note for any $\lambda' < \lambda$,
\begin{align*}
  \lambda' P\left( \sup_{i\in I, i\leq M}X_i>\lambda' \right)
  &= \lambda' P\left( \bigcup_{i\in I, i\leq M}\{X_i>\lambda'\} \right) \\
  &= \lambda' \lim_{n\to\infty} P\left( \bigcup_{i\in J_n, i\leq M}\{X_i>\lambda'\} \right) \\
  &\le \lim_{n\to\infty} \mathbb{E}\left[X_{M} \mathbb{I}_{\sup_{i \in J_n, i \le M}X_i \ge \lambda'}\right] \\
  &\le \mathbb{E}\left[X_{M} \mathbb{I}_{\sup_{i \in I, i \le M}X_i \ge \lambda'}\right]
\end{align*}
Now take $\lambda'\uparrow\lambda$ and use monotone or dominated convergence to derive the desired inequality.

In the above, we first interpret $\sup_{i\in I, i\leq M}X_i$ in extended reals for well-definedness,
then observe by the (now proved) maximal inequality that $\sup_{i\in I, i\leq M}X_i < \infty$ a.s.\ by taking $\lambda\to\infty$,
so the supremum is well-defined in $\mathbb{R}$.
%See 8.1.1 Pascucci.
\end{proof}


\begin{lemma}[Doob Lp Inequality for countable]\label{lem:doob_Lp_countable}
  Let $X : I \rightarrow \Omega \rightarrow \mathbb{R}$ be a non-negative sub-martingale. Let $I$ be countable.
  For every $M\in I,\lambda > 0$ and $p>1$ we have
  \begin{align*}
    \mathbb{E}\left[ \sup_{i\in I, i \leq M}X_i^p \right]
    \leq \left(\frac{p}{p-1}\right)^p\mathbb{E}[X_M^p]
    \: .
  \end{align*}
  That is, for $\Vert \cdot \Vert_p$ the $L^p$ norm,
  $\left\Vert \sup_{i \le M}  X_i  \right\Vert_p
    \leq \frac{p}{p-1} \left\Vert X_M \right\Vert_p
    \: .$
\end{lemma}

\begin{proof}
  \uses{lem:doob_countable}
\begin{align*}
  \mathbb{E}\left[ \sup_{i \le M}X_i^p \right]
  = p \int_0^\infty \mathbb{P}\left( \sup_{i \le M}X_i \geq \lambda \right) \lambda^{p-1} d\lambda
\end{align*}
By Theorem~\ref{lem:doob_countable} and then Fubini's theorem, we have then
\begin{align*}
  \mathbb{E}\left[ \sup_{i \le M}X_i^p \right]
  &\le p \int_0^\infty \mathbb{E}\left[X_M \mathbb{I}_{\sup_{i \le M}X_i \ge \lambda}\right] \lambda^{p-2} d\lambda
  \\
  &= p \mathbb{E}\left[X_M \int_0^{\sup_{i \le M}X_i} \lambda^{p-2} d\lambda\right]
  \\
  &= \frac{p}{p - 1} \mathbb{E}\left[X_M (\sup_{i \le M}X_i)^{p-1}\right]
  \: .
\end{align*}
Then by Hölder's inequality,
\begin{align*}
  \mathbb{E}\left[ \sup_{i \le M}X_i^p \right]
  &\le \frac{p}{p - 1} \left(\mathbb{E}[X_M^p]\right)^{1/p} \left(\mathbb{E}\left[\sup_{i \le M}X_i^p \right]\right)^{(p-1)/p}
  \: .
\end{align*}
We then divide the two sides by $\left(\mathbb{E}\left[\sup_{i \le M}X_i^p \right]\right)^{(p-1)/p}$ and raise to the power $p$ to conclude.
%  8.1.1 Pascucci.
\end{proof}


\begin{theorem}[Doob Inequality]\label{thm:doob_ineq}
  \uses{def:Submartingale, def:RightContinuous}
  \leanok
  \lean{ProbabilityTheory.maximal_ineq}
  Let $X: \mathbb{R}_+ \to \Omega \to \mathbb{R}$ be a right-continuous non-negative sub-martingale.
  For every $T \in \mathbb{R}_+$ and $\lambda>0$ we have
  \begin{align*}
    P\left( \sup_{t\in[0,T]}X_t \geq \lambda \right)
    \leq \frac{\mathbb{E}[X_T \mathbb{I}_{\sup_{i \le T}X_i \ge \lambda}]}{\lambda}
    \leq \frac{\mathbb{E}[X_T]}{\lambda}
    \: .
  \end{align*}
\end{theorem}

\begin{proof}
  \uses{lem:doob_countable}
Since $X$ is right-continuous and $[0,T]$ is a compact interval, we have that
\begin{align*}
  \sup_{t\in[0,T]}X_t = \sup_{t\in[0,T] \cap \mathbb{Q}}X_t
  \: .
\end{align*}
Then apply Lemma~\ref{lem:doob_countable} with $I = [0,T] \cap \mathbb{Q}$ and $M = T$.
\end{proof}

\begin{lemma}\label{lem:lintegral_rpow_le_of_forall_mul_meas_le}
  \leanok
  \lean{MeasureTheory.lintegral_rpow_le_of_forall_mul_meas_le}
  Let $\mu$ be a measure and let $f, g$ be measurable functions with values in $[0, \infty]$ such that for all $\lambda > 0$, $\mu(f \ge \lambda) < \infty$ and
  $\lambda \mu(f \ge \lambda) \le \mu[g \mathbb{I}_{f \ge \lambda}]$.
  Then for every $p > 1$,
  \begin{align*}
    \mu[f^p] \le \left(\frac{p}{p-1}\right)^p \mu[g^p]
    \: .
  \end{align*}
\end{lemma}

\begin{proof}
  \leanok
We first reduce to the case of a finite measure. For $\delta > 0$, the restriction $\mu_\delta$ of $\mu$ to $\{f \ge \delta\}$ is a finite measure, and it satisfies the same assumption: for $\lambda \ge \delta$ this is the assumption for $\mu$, and for $\lambda < \delta$ we have $\lambda \mu_\delta(f \ge \lambda) = \lambda \mu(f \ge \delta) \le \delta \mu(f \ge \delta) \le \mu[g \mathbb{I}_{f \ge \delta}] = \mu_\delta[g \mathbb{I}_{f \ge \lambda}]$.
The inequality for $\mu_\delta$ gives $\mu[f^p \mathbb{I}_{f \ge \delta}] \le \left(\frac{p}{p-1}\right)^p \mu[g^p]$, and we let $\delta$ tend to $0$ by monotone convergence.

Suppose now that $\mu$ is finite.
By monotone convergence it suffices to prove the inequality for $f_n = f \wedge n$ in place of $f$. Note that $\{f_n \ge \lambda\}$ is $\{f \ge \lambda\}$ for $\lambda \le n$ and is empty otherwise.
By the layer cake formula applied to $\mu$, then to the measure $g \cdot \mu$,
\begin{align*}
  \mu[f_n^p]
  = p \int_0^\infty \mu(f_n \ge \lambda) \lambda^{p-1} d\lambda
  \le p \int_0^\infty (g \cdot \mu)(f_n \ge \lambda) \lambda^{p-2} d\lambda
  = \frac{p}{p - 1} \mu\left[g f_n^{p-1}\right]
  \: .
\end{align*}
Then by H\"older's inequality,
$\mu[f_n^p] \le \frac{p}{p - 1} \left(\mu[g^p]\right)^{1/p} \left(\mu[f_n^p]\right)^{(p-1)/p}$.
Since $\mu[f_n^p]$ is finite, we can divide the two sides by $\left(\mu[f_n^p]\right)^{(p-1)/p}$ and raise to the power $p$ to conclude.
\end{proof}


\begin{theorem}[Doob's Lp inequality in $\mathbb{R}$]\label{thm:doob_lp}
  \uses{def:Submartingale, def:RightContinuous}
  \leanok
  \lean{ProbabilityTheory.lintegral_iSup_le_ofReal_rpow_le}
  Let $X:\mathbb{R} \rightarrow \Omega \rightarrow \mathbb{R}$ be a right-continuous non-negative sub-martingale.
  For every $T, \lambda>0$ and $p>1$ we have
  \begin{align*}
    \mathbb{E}\left[ \sup_{t\in[0,T]}X_t^p \right]
    \leq \left(\frac{p}{p-1}\right)^p\mathbb{E}[X_T^p]
    \: .
  \end{align*}
  That is, for $\Vert \cdot \Vert_p$ the $L^p$ norm,
  $\left\Vert \sup_{t\in[0,T]}  X_t  \right\Vert_p
    \leq \frac{p}{p-1} \left\Vert X_T \right\Vert_p
    \: .$
\end{theorem}

\begin{proof}
  \leanok
  \uses{thm:doob_ineq, lem:lintegral_rpow_le_of_forall_mul_meas_le}
  By Theorem~\ref{thm:doob_ineq}, the functions $f = \sup_{t \in [0,T]} X_t$ and $g = X_T$ satisfy the assumption of Lemma~\ref{lem:lintegral_rpow_le_of_forall_mul_meas_le}.
\end{proof}


\begin{corollary}\label{cor:doob_lp_top}
  \uses{def:Submartingale, def:RightContinuous}
  \leanok
  \lean{ProbabilityTheory.lintegral_iSup_ofReal_rpow_le}
  Let $X:\mathbb{R} \rightarrow \Omega \rightarrow \mathbb{R}$ be a right-continuous non-negative sub-martingale.
  For every $p>1$ we have
  \begin{align*}
    \mathbb{E}\left[ \sup_t X_t^p \right]
    \leq \left(\frac{p}{p-1}\right)^p \sup_t \mathbb{E}[X_t^p]
    \: .
  \end{align*}
\end{corollary}

\begin{proof}
  \leanok
  \uses{thm:doob_lp}
  Let $(t_n)$ be a nondecreasing sequence of times which tends to infinity. By Theorem~\ref{thm:doob_lp},
  $\mathbb{E}\left[ \sup_{t \le t_n} X_t^p \right] \le \left(\frac{p}{p-1}\right)^p \sup_t \mathbb{E}[X_t^p]$ for all $n$, and we conclude by monotone convergence.
\end{proof}
